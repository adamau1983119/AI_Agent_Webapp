# 程式編寫邏輯與介面設計 - 階段 1

## 📋 概述

本文檔說明階段 1 的程式編寫邏輯，重點是建立前端與後台的溝通介面，讓客戶可以通過文字或照片內容調整後台的搜尋方向。

---

## 🎯 核心需求

### 客戶重點要求

**關鍵功能**：客戶需要能夠通過前端介面輸入文字或上傳照片，這些輸入會影響後台的搜尋方向，從而生成更符合客戶需求的內容。

**具體場景**：
1. 客戶在主題列表看到一個標題，點擊展開
2. 客戶可以：
   - 輸入文字描述（如「強調材質與設計細節」）
   - 上傳參考照片（如 popbee 風格的照片）
   - 選擇風格偏好（如「專業記者風格」）
3. 這些輸入會傳遞給後台，調整：
   - 圖片搜尋關鍵字
   - 文章生成重點
   - 內容風格

**邏輯流程**：
```python
async def search_images(
    self,
    keywords: List[str],
    count: int = 8,
    mode: str = "full",  # "preview" 或 "full"
    customer_input: Optional[Dict[str, Any]] = None  # 客戶輸入
) -> List[Dict[str, Any]]:
    """搜尋圖片"""
    
    # 1. 根據模式調整搜尋策略
    if mode == "preview":
        # Preview 模式：只搜尋 1 張高級感圖片
        return await self._search_preview_images(keywords, customer_input)
    else:
        # Full 模式：搜尋 8 張相關圖片
        return await self._search_full_images(keywords, customer_input)

async def _search_preview_images(
    self,
    keywords: List[str],
    customer_input: Optional[Dict[str, Any]] = None
) -> List[Dict[str, Any]]:
    """Preview 模式：搜尋高級感圖片"""
    
    # 1. 關鍵字優化（加入高級感相關關鍵字）
    enhanced_keywords = self._enhance_keywords_for_preview(keywords)
    
    # 2. 如果客戶有輸入，調整搜尋方向
    if customer_input:
        enhanced_keywords = self._adjust_keywords_from_customer_input(
            enhanced_keywords,
            customer_input
        )
    
    # 3. 搜尋圖片（優先使用高品質服務）
    images = await self._search_with_quality_filter(
        enhanced_keywords,
        count=1,
        min_resolution="1920x1080"
    )
    
    return images

def _enhance_keywords_for_preview(self, keywords: List[str]) -> List[str]:
    """為 Preview 模式優化關鍵字"""
    # 從配置檔讀取關鍵字增強列表
    config = TopicGenerationConfig()
    enhancement_keywords = config.get_preview_keywords_enhancement()
    
    # 合併關鍵字
    enhanced = keywords + enhancement_keywords
    return enhanced

def _adjust_keywords_from_customer_input(
    self,
    keywords: List[str],
    customer_input: Dict[str, Any]
) -> List[str]:
    """根據客戶輸入調整關鍵字"""
    adjusted = keywords.copy()
    
    # 1. 如果客戶有文字描述，提取關鍵字
    if customer_input.get('reference_text'):
        text_keywords = self._extract_keywords_from_text(
            customer_input['reference_text']
        )
        adjusted.extend(text_keywords)
    
    # 2. 如果客戶有照片，提取視覺特徵關鍵字
    if customer_input.get('reference_image_url'):
        image_keywords = await self._extract_keywords_from_image(
            customer_input['reference_image_url']
        )
        adjusted.extend(image_keywords)
    
    # 3. 如果客戶有重點提示，加入相關關鍵字
    if customer_input.get('focus'):
        focus_keywords = self._map_focus_to_keywords(
            customer_input['focus']
        )
        adjusted.extend(focus_keywords)
    
    return adjusted
```

#### 後台處理邏輯

**檔案位置**：`backend/app/api/v1/topics.py`

```python
@router.post("/{topic_id}/expand")
async def expand_topic(
    topic_id: str = Path(..., description="主題 ID"),
    request: ExpandTopicRequest = Body(...)
):
    """展開主題（生成完整內容）"""
    
    # 1. 檢查主題是否存在
    topic = await topic_repo.get_topic_by_id(topic_id)
    if not topic:
        raise HTTPException(404, "主題不存在")
    
    # 2. 檢查是否已展開（使用快取）
    if topic.is_expanded:
        cached_content = await cache_service.get_expanded_content(topic_id)
        if cached_content:
            return cached_content
    
    # 3. 解析客戶輸入，調整搜尋策略
    search_strategy = await _build_search_strategy_from_customer_input(request)
    
    # 4. 生成文章（根據 style + focus）
    article = await content_service.generate_article(
        topic_id=topic_id,
        style=request.style or "popbee_professional",
        focus=request.focus,
        reference_text=request.reference_text
    )
    
    # 5. 搜尋圖片（根據關鍵字 + 客戶輸入）
    images = await image_service.search_images(
        keywords=article.extracted_keywords,
        count=8,
        mode="full",
        customer_input={
            "reference_text": request.reference_text,
            "reference_image_url": request.reference_image_url,
            "image_annotations": request.image_annotations,
            "focus": request.focus
        }
    )
    
    # 6. 儲存結果並更新主題
    expanded_content = {
        "article": article,
        "images": images
    }
    
    # 快取結果
    await cache_service.set_expanded_content(topic_id, expanded_content, ttl=86400)
    
    # 更新主題
    await topic_repo.update_topic(topic_id, {
        "is_expanded": True,
        "customer_input": request.dict(),
        "style": request.style,
        "focus": request.focus,
        "reference_text": request.reference_text,
        "reference_image_url": request.reference_image_url
    })
    
    return expanded_content

async def _build_search_strategy_from_customer_input(
    request: ExpandTopicRequest
) -> Dict[str, Any]:
    """根據客戶輸入建立搜尋策略"""
    strategy = {
        "keywords": [],
        "image_style": "editorial_fashion",  # 預設
        "quality_requirements": {
            "min_resolution": "1920x1080",
            "min_saturation": 0.6
        }
    }
    
    # 1. 從文字描述提取關鍵字
    if request.reference_text:
        text_keywords = await nlp_service.extract_keywords(
            request.reference_text
        )
        strategy["keywords"].extend(text_keywords)
    
    # 2. 從照片提取視覺特徵
    if request.reference_image_url:
        image_features = await image_analysis_service.extract_features(
            request.reference_image_url
        )
        strategy["keywords"].extend(image_features["keywords"])
        strategy["image_style"] = image_features.get("style", "editorial_fashion")
    
    # 3. 從照片標註提取關鍵字
    if request.image_annotations:
        annotations = request.image_annotations
        if annotations.get("theme"):
            strategy["keywords"].append(annotations["theme"])
        if annotations.get("elements"):
            strategy["keywords"].extend(annotations["elements"])
        if annotations.get("feeling"):
            strategy["keywords"].append(annotations["feeling"])
    
    # 4. 從重點提示映射關鍵字
    if request.focus:
        focus_keywords = await _map_focus_to_keywords(request.focus)
        strategy["keywords"].extend(focus_keywords)
    
    return strategy
```

async def extract_keywords_from_text(text: str) -> List[str]:
    """從客戶文字描述提取關鍵字"""
    # 使用 NLP 服務提取關鍵字
    keywords = await nlp_service.extract_keywords(text)
    
    # 過濾和優化關鍵字
    filtered = [
        kw for kw in keywords 
        if len(kw) > 2 and kw not in STOP_WORDS
    ]
    
    return filtered

def map_annotations_to_keywords(annotations: Dict[str, Any]) -> List[str]:
    """將照片標註映射為搜尋關鍵字"""
    keywords = []
    
    # 主題映射
    theme_mapping = {
        "人物": ["portrait", "person", "model"],
        "場景": ["scene", "setting", "background"],
        "物品": ["product", "object", "item"]
    }
    if annotations.get("theme"):
        keywords.extend(theme_mapping.get(annotations["theme"], []))
    
    # 元素映射
    if annotations.get("elements"):
        keywords.extend(annotations["elements"])
    
    # 感覺映射
    feeling_mapping = {
        "高級感": ["luxury", "premium", "high-end"],
        "街頭感": ["street", "urban", "casual"],
        "科技感": ["tech", "modern", "futuristic"]
    }
    if annotations.get("feeling"):
        keywords.extend(feeling_mapping.get(annotations["feeling"], []))
    
    return keywords
```

    """根據重點提示調整搜尋策略"""

    return strategy
```

```python
def merge_search_strategies(
    base_strategy: Dict[str, Any],
    customer_input: Dict[str, Any]
) -> Dict[str, Any]:
    """合併基礎策略與客戶輸入"""
    merged = base_strategy.copy()
    
    # 客戶輸入的關鍵字優先
    if customer_input.get("keywords"):
        merged["keywords"] = customer_input["keywords"] + merged["keywords"]
    
    # 客戶輸入的風格優先
    if customer_input.get("image_style"):
        merged["image_style"] = customer_input["image_style"]
    
    # 客戶輸入的品質要求優先
    if customer_input.get("quality_requirements"):
        merged["quality_requirements"].update(
            customer_input["quality_requirements"]
        )
    
    return merged
```
