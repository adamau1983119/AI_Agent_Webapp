## 🔌 核心 API 設計

### API 端點：`POST /api/v1/topics/{id}/expand`

**請求格式**（使用 Pydantic 驗證）：
```json
{
  "style": "popbee_professional",           // 可選：風格選擇（必須在允許清單中）
  "focus": "材質與設計細節",                 // 可選：重點提示（必須在允許清單中）
  "reference_text": "客戶輸入的文字描述",     // 可選：範例文字（最大長度限制）
  "reference_image_url": "照片URL",          // 可選：參考照片（必須驗證格式與大小）
  "image_annotations": {                     // 可選：照片標註（必須符合 schema）
    "theme": "人物",
    "elements": ["鞋款", "配色"],
    "feeling": "高級感"
  }
}
```

**Pydantic Schema 定義**（`backend/app/schemas/topic.py`）：
```python
from pydantic import BaseModel, Field, validator, HttpUrl
from typing import List, Optional, Literal

class ImageAnnotations(BaseModel):
    """圖片標註 Schema"""
    theme: Optional[Literal["人物", "場景", "物品"]] = None
    elements: Optional[List[str]] = Field(default_factory=list, max_items=10)
    feeling: Optional[Literal["高級感", "街頭感", "科技感"]] = None

class ExpandTopicRequest(BaseModel):
    """展開主題請求 Schema（含輸入驗證）"""
    style: Optional[Literal["popbee_professional", "casual_lifestyle", "tech_analysis"]] = None
    focus: Optional[Literal["材質與設計細節", "品牌背景", "流行趨勢分析"]] = None
    reference_text: Optional[str] = Field(None, max_length=5000, description="最大長度 5000 字")
    reference_image_url: Optional[HttpUrl] = Field(None, description="必須是有效的 HTTP/HTTPS URL")
    image_annotations: Optional[ImageAnnotations] = None
    
    @validator('reference_image_url')
    def validate_image_url(cls, v):
        """驗證圖片 URL 格式與大小"""
        if v:
            # 檢查 URL 格式
            if not str(v).startswith(('http://', 'https://')):
                raise ValueError('圖片 URL 必須是 HTTP/HTTPS')
            # 檢查檔案大小（通過 HEAD 請求）
            # 實際實作時需要檢查檔案大小限制（例如 ≤5MB）
        return v
```

```python
@router.post("/{topic_id}/expand")
async def expand_topic(
    topic_id: str = Path(..., description="主題 ID"),
    request: ExpandTopicRequest = Body(...)  # Pydantic 自動驗證
):
    # 1. 檢查快取（如果相同輸入已生成過）
    cache_key = f"topic:{topic_id}:expand:{hash(request.json())}"
    cached_result = await cache_service.get(cache_key)
    if cached_result:
        return {
            **cached_result,
            "cache_hit": True  # 標記使用快取
        }
    
    # 2. 解析客戶輸入，建立搜尋策略（從配置檔讀取）
    search_strategy = await _build_search_strategy_from_config(request)
    
    # 3. 生成文章（根據 style + focus）
    article = await content_service.generate_article(
        topic_id=topic_id,
        style=request.style,
        focus=request.focus,
        reference_text=request.reference_text
    )
    
    # 4. 搜尋圖片（根據客戶輸入調整，含容錯機制）
    images = await image_service.search_images(
        keywords=article.extracted_keywords,
        count=8,
        mode="full",
        customer_input=request.dict(),
        fallback_enabled=True  # 啟用容錯機制
    )
    
    result = {
        "article": article,
        "images": images,
        "cache_hit": False,
        "generation_time": time.time() - start_time
    }
    
    # 5. 儲存快取（TTL 24 小時）
    await cache_service.set(cache_key, result, ttl=86400)
    
    return result
```

# backend/app/services/images/keyword_mapper.py
class KeywordMapper:
    def __init__(self):
        # 從配置檔讀取映射規則
        self.config = TopicGenerationConfig()
        self.theme_map = self.config.get_theme_mapping()
        self.feeling_map = self.config.get_feeling_mapping()
    
    def map_annotations_to_keywords(self, annotations: Dict) -> List[str]:
        """將照片標註映射為搜尋關鍵字（從配置檔讀取）"""
        keywords = []
        
        # 主題映射（從配置檔）
        if annotations.get("theme") and annotations["theme"] in self.theme_map:
            keywords.extend(self.theme_map[annotations["theme"]])
        
        # 感覺映射（從配置檔）
        if annotations.get("feeling") and annotations["feeling"] in self.feeling_map:
            keywords.extend(self.feeling_map[annotations["feeling"]])
        
        # 元素直接加入
        if annotations.get("elements"):
            keywords.extend(annotations["elements"])
        
        return keywords
```

class StrategyAdjuster:
    def __init__(self):
        # 從配置檔讀取策略
        self.config = TopicGenerationConfig()
        self.focus_map = self.config.get_focus_mapping()
    
    def adjust_strategy_by_focus(self, focus: str, base_strategy: Dict) -> Dict:
        """根據重點調整搜尋策略（從配置檔讀取）"""
        if focus and focus in self.focus_map:
            config = self.focus_map[focus]
            base_strategy["keywords"].extend(config["keywords"])
            base_strategy["image_style"] = config["image_style"]
            base_strategy["min_resolution"] = config.get("min_resolution", "1920x1080")
        
        return base_strategy
```

### 圖片搜尋服務（`backend/app/services/images/image_service.py`）

```python
async def search_images(
    self,
    keywords: List[str],
    count: int = 8,
    mode: str = "full",
    customer_input: Optional[Dict] = None,
    fallback_enabled: bool = True  # 容錯機制
) -> List[Dict]:
    """搜尋圖片（根據客戶輸入調整，含容錯機制）"""
    
    # 1. 基礎關鍵字
    search_keywords = keywords.copy()
    
    # 2. 根據客戶輸入調整關鍵字
    if customer_input:
        # 從文字描述提取關鍵字
        if customer_input.get("reference_text"):
            text_keywords = await extract_keywords_from_text(
                customer_input["reference_text"]
            )
            search_keywords.extend(text_keywords)
        
        # 從照片提取視覺特徵
        if customer_input.get("reference_image_url"):
            try:
                image_features = await extract_features_from_image(
                    customer_input["reference_image_url"]
                )
                search_keywords.extend(image_features["keywords"])
            except Exception as e:
                logger.warning(f"圖片分析失敗: {e}")  # 記錄但不中斷流程
        
        # 從照片標註映射關鍵字
        if customer_input.get("image_annotations"):
            mapper = KeywordMapper()
            annotation_keywords = mapper.map_annotations_to_keywords(
                customer_input["image_annotations"]
            )
            search_keywords.extend(annotation_keywords)
        
        # 根據重點調整策略（從配置檔讀取）
        if customer_input.get("focus"):
            adjuster = StrategyAdjuster()
            search_strategy = adjuster.adjust_strategy_by_focus(
                customer_input["focus"],
                {"keywords": search_keywords}
            )
            search_keywords = search_strategy["keywords"]
    
    # 3. 去重並優化關鍵字
    search_keywords = list(set(search_keywords))
    
    # 4. 執行搜尋（含容錯機制）
    images = await self._search_with_fallback(
        keywords=search_keywords,
        count=count,
        quality_filter=True,
        fallback_enabled=fallback_enabled
    )
    
    return images

async def _search_with_fallback(
    self,
    keywords: List[str],
    count: int,
    quality_filter: bool,
    fallback_enabled: bool
) -> List[Dict]:
    """使用容錯機制搜尋圖片"""
    config = TopicGenerationConfig()
    service_priority = config.get_image_service_priority()
    images = []
    errors = []
    
    # 按優先順序嘗試每個服務
    for service_name in service_priority:
        try:
            service = self.service_manager.get_service(service_name)
            service_images = await service.search(
                keywords=keywords,
                count=count - len(images),
                quality_filter=quality_filter
            )
            
            # 為每張圖片標記來源
            for img in service_images:
                img["source"] = service_name
            
            images.extend(service_images)
            
            # 如果已收集足夠圖片，停止搜尋
            if len(images) >= count:
                break
                
        except Exception as e:
            # 記錄錯誤但不中斷
            error_msg = f"{service_name} 搜尋失敗: {e}"
            logger.warning(error_msg)
            errors.append(error_msg)
            
            # 如果啟用容錯，繼續嘗試下一個服務
            if not fallback_enabled:
                raise
    
    # 如果所有服務都失敗，記錄錯誤
    if len(images) == 0 and errors:
        logger.error(f"所有圖片服務都失敗: {errors}")
        # 可以選擇返回空列表或拋出異常
    
    return images[:count]  # 返回指定數量
```

## 💾 快取服務設計

### 快取服務（`backend/app/services/cache/cache_service.py`）

```python
from typing import Optional, Any
import hashlib
import json
import redis  # 或使用 MongoDB

class CacheService:
    """快取服務（支援 Redis 或 MongoDB）"""
    
    def __init__(self):
        self.backend = self._init_backend()  # Redis 或 MongoDB
        self.default_ttl = 86400  # 24 小時
    
    async def get(self, key: str) -> Optional[Any]:
        """取得快取"""
        try:
            if isinstance(self.backend, redis.Redis):
                data = self.backend.get(key)
                return json.loads(data) if data else None
            else:  # MongoDB
                doc = await self.backend.find_one({"key": key})
                return doc["value"] if doc else None
        except Exception as e:
            logger.warning(f"快取讀取失敗: {e}")
            return None
    
    async def set(self, key: str, value: Any, ttl: Optional[int] = None):
        """設定快取"""
        try:
            ttl = ttl or self.default_ttl
            if isinstance(self.backend, redis.Redis):
                self.backend.setex(key, ttl, json.dumps(value))
            else:  # MongoDB
                await self.backend.update_one(
                    {"key": key},
                    {"$set": {"value": value, "expires_at": datetime.utcnow() + timedelta(seconds=ttl)}},
                    upsert=True
                )
        except Exception as e:
            logger.warning(f"快取寫入失敗: {e}")
    
    def generate_key(self, topic_id: str, customer_input: Dict) -> str:
        """生成快取 Key（基於 topic_id + 客戶輸入的 hash）"""
        input_hash = hashlib.md5(
            json.dumps(customer_input, sort_keys=True).encode()
        ).hexdigest()
        return f"topic:{topic_id}:expand:{input_hash}"
```
