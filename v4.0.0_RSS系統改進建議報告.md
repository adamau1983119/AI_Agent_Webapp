### 改進建議 2.1：動態角色分配

**建議實作**：

```python
# backend/app/services/dynamic_role_allocation.py

class DynamicRoleAllocator:
    """動態角色分配器"""
    
    def __init__(self, health_service: FeedHealthService):
        self.health_service = health_service
    
    async def get_role_distribution(
        self,
        category: Category,
        total_count: int = 10
    ) -> Dict[str, int]:
        """
        根據來源健康度動態分配角色比例
        
        規則：
        1. 健康來源多的角色，分配更多配額
        2. 失敗率高的角色，降低配額
        3. 確保至少 1 個配額/角色（避免完全失效）
        """
        roles = get_roles_for_category(category)
        role_health_scores = {}
        
        # 計算每個角色的健康分數
        for role_name, feeds in roles.items():
            scores = []
            for source_name, feed_url, _ in feeds:
                health = await self.health_service.get_feed_health(feed_url)
                if health:
                    scores.append(health.get("success_rate", 0.5))
                else:
                    scores.append(0.7)  # 新來源預設 0.7
            
            role_health_scores[role_name] = sum(scores) / len(scores) if scores else 0.5
        
        # 根據健康分數分配配額
        total_score = sum(role_health_scores.values())
        distribution = {}
        
        for role_name, score in role_health_scores.items():
            # 基礎配額 + 健康加權
            base_quota = 1
            weighted_quota = int((score / total_score) * (total_count - len(roles)))
            distribution[role_name] = base_quota + weighted_quota
        
        # 調整確保總數正確
        current_total = sum(distribution.values())
        if current_total != total_count:
            diff = total_count - current_total
            # 將差額分配給最健康的角色
            best_role = max(role_health_scores, key=role_health_scores.get)
            distribution[best_role] += diff
        
        return distribution
```

**預期效果**：
- 健康來源利用率提升 30%
- 減少因來源失效導致的空內容

### 改進建議 3.2：新增 RSS 自動驗證機制

**建議實作**：

```python
# backend/app/services/rss_validator.py

class RSSValidator:
    """RSS 來源自動驗證器"""
    
    async def validate_all_feeds(self) -> Dict[str, Any]:
        """
        每日自動驗證所有 RSS 來源
        
        Returns:
            {
                "total": 150,
                "valid": 140,
                "invalid": 10,
                "invalid_feeds": [...],
                "suggestions": [...]
            }
        """
        results = {
            "total": 0,
            "valid": 0,
            "invalid": 0,
            "invalid_feeds": [],
            "suggestions": []
        }
        
        all_feeds = self._get_all_configured_feeds()
        
        async with httpx.AsyncClient(timeout=10.0) as client:
            for feed in all_feeds:
                results["total"] += 1
                
                try:
                    response = await client.get(feed["url"])
                    
                    if response.status_code == 200:
                        # 驗證是否為有效 RSS
                        parsed = feedparser.parse(response.text)
                        if parsed.entries and len(parsed.entries) > 0:
                            results["valid"] += 1
                        else:
                            results["invalid"] += 1
                            results["invalid_feeds"].append({
                                "name": feed["name"],
                                "url": feed["url"],
                                "reason": "Empty feed",
                                "suggestion": self._find_alternative(feed)
                            })
                    else:
                        results["invalid"] += 1
                        results["invalid_feeds"].append({
                            "name": feed["name"],
                            "url": feed["url"],
                            "reason": f"HTTP {response.status_code}",
                            "suggestion": self._find_alternative(feed)
                        })
                        
                except Exception as e:
                    results["invalid"] += 1
                    results["invalid_feeds"].append({
                        "name": feed["name"],
                        "url": feed["url"],
                        "reason": str(e),
                        "suggestion": self._find_alternative(feed)
                    })
        
        return results
    
    def _find_alternative(self, failed_feed: Dict) -> Optional[str]:
        """
        從備用池中尋找替代來源
        """
        # 從 BACKUP_FEEDS 中尋找同類別的替代
        category = failed_feed.get("category")
        role = failed_feed.get("role")
        
        alternatives = BACKUP_FEEDS.get(category, {}).get(role, [])
        for alt in alternatives:
            if alt["url"] != failed_feed["url"]:
                return alt["url"]
        
        return None
```

### 改進建議 4.1：實作內容去重演算法

**建議實作**：

```python
# backend/app/services/content_deduplication.py

import hashlib
from difflib import SequenceMatcher
from typing import List, Set, Optional
from datetime import datetime, timedelta

class ContentDeduplicator:
    """內容去重服務"""
    
    def __init__(self, db):
        self.db = db
        self.title_cache: Set[str] = set()
        self.recent_titles: List[str] = []
        self.cache_ttl_hours = 24
        
    async def initialize_cache(self):
        """從資料庫載入最近 24 小時的標題"""
        cutoff = datetime.utcnow() - timedelta(hours=self.cache_ttl_hours)
        
        recent_topics = await self.db.topics.find(
            {"created_at": {"$gte": cutoff}},
            {"title": 1}
        ).to_list(length=10000)
        
        for topic in recent_topics:
            title = topic.get("title", "")
            self.title_cache.add(self._hash_title(title))
            self.recent_titles.append(title)
    
    def _hash_title(self, title: str) -> str:
        """生成標題哈希"""
        normalized = title.lower().strip()
        return hashlib.md5(normalized.encode()).hexdigest()
    
    def _compute_similarity(self, title1: str, title2: str) -> float:
        """計算兩個標題的相似度"""
        return SequenceMatcher(None, title1.lower(), title2.lower()).ratio()
    
    async def is_duplicate(
        self,
        title: str,
        similarity_threshold: float = 0.85
    ) -> tuple[bool, Optional[str]]:
        """
        檢查標題是否重複
        
        Returns:
            (is_duplicate, reason)
        """
        # 方法 1：精確匹配（哈希）
        title_hash = self._hash_title(title)
        if title_hash in self.title_cache:
            return True, "exact_match"
        
        # 方法 2：模糊匹配（相似度）
        for existing_title in self.recent_titles[-500:]:  # 只檢查最近 500 條
            similarity = self._compute_similarity(title, existing_title)
            if similarity > similarity_threshold:
                return True, f"similar_to: {existing_title[:30]}... ({similarity:.0%})"
        
        # 不重複，加入快取
        self.title_cache.add(title_hash)
        self.recent_titles.append(title)
        
        # 控制快取大小
        if len(self.recent_titles) > 10000:
            self.recent_titles = self.recent_titles[-5000:]
        
        return False, None
    
    async def get_duplicate_stats(self) -> Dict[str, Any]:
        """獲取去重統計"""
        return {
            "cache_size": len(self.title_cache),
            "recent_titles_count": len(self.recent_titles),
            "cache_ttl_hours": self.cache_ttl_hours
        }
```

```python
# backend/app/services/keyword_updater.py

class DynamicKeywordUpdater:
    """動態關鍵字更新器"""
    
    async def update_fallback_keywords(self, category: Category):
        """
        每週從成功收集的文章中提取熱門關鍵字
        更新 Layer 3 的備用關鍵字庫
        """
        # 獲取過去 7 天的熱門文章
        cutoff = datetime.utcnow() - timedelta(days=7)
        
        recent_topics = await self.db.topics.find({
            "category": category.value,
            "created_at": {"$gte": cutoff},
            "score": {"$gte": 0.7}  # 只取高分文章
        }).to_list(length=100)
        
        # 提取關鍵字
        all_keywords = []
        for topic in recent_topics:
            keywords = topic.get("hashtags", [])
            all_keywords.extend(keywords)
        
        # 統計詞頻
        from collections import Counter
        keyword_counts = Counter(all_keywords)
        
        # 取 Top 20
        top_keywords = [kw for kw, count in keyword_counts.most_common(20)]
        
        # 更新到資料庫
        await self.db.system_config.update_one(
            {"key": f"fallback_keywords_{category.value}"},
            {"$set": {
                "keywords": top_keywords,
                "updated_at": datetime.utcnow()
            }},
            upsert=True
        )
        
        logger.info(f"更新 {category.value} 備用關鍵字: {top_keywords[:5]}...")
        
        return top_keywords
```

## 🟡 問題 6：資料保留策略過於簡單

### 現況

```yaml
data_cleanup:
  retention_days: 15  # 一刀切
```

### 改進建議 6.1：分層資料保留

**建議調整**：

```yaml
# backend/config/topic_generation.yaml

data_cleanup:
  enabled: true
  
  # 分層保留策略
  retention_tiers:
    # Tier 1: 熱門內容（高互動）
    hot:
      condition: "view_count >= 100 OR rating_count >= 10"
      retention_days: 60
      
    # Tier 2: 優質內容（高評分）
    quality:
      condition: "score >= 0.8"
      retention_days: 45
      
    # Tier 3: 一般內容
    normal:
      condition: "score >= 0.5"
      retention_days: 30
      
    # Tier 4: 低品質內容
    low:
      condition: "score < 0.5"
      retention_days: 15
      
    # Tier 5: AI 生成內容
    ai_generated:
      condition: "ai_generated == true"
      retention_days: 7  # AI 內容保留較短
  
  # 清理排程
  cleanup_schedule:
    hour: 3
    minute: 0
    batch_size: 100
```
